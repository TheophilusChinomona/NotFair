# E2E Testing Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Playwright E2E tests for all major app flows and wire them into a GitHub Actions CI workflow.

**Architecture:** Auth bypass via `E2E_TEST=1` env var in middleware. Page object model under `e2e/pages/`. Tests under `e2e/specs/`. CI runs on GitHub-hosted runners with SQLite (no external infra).

**Tech Stack:** Playwright, Chromium, GitHub Actions, pnpm, SQLite

## Global Constraints

- `E2E_TEST=1` env var bypasses auth in middleware only
- All tests run against a single shared app instance on `http://localhost:3326`
- SQLite default DB — no external database required
- Chromium only (single browser)
- `@playwright/test` as dev dependency
- Report upload on CI failure only

import { type Page, type Locator } from "@playwright/test";

export class DashboardPage {
  readonly page: Page;
  readonly sidebar: Locator;
  readonly projectName: Locator;

  constructor(page: Page) {
    this.page = page;
    this.sidebar = page.locator("nav, [role=navigation], aside");
    this.projectName = page.locator("h1, [data-testid=project-name]");
  }

  async goto() {
    await this.page.goto("/");
  }

  async sidebarLinks() {
    return this.sidebar.locator("a");
  }

  /** Navigate via sidebar link containing `label` text */
  async navigateTo(label: string) {
    await this.sidebar.locator(`a:has-text("${label}")`).first().click();
    await this.page.waitForLoadState("networkidle");
  }
}

import { type Page, type Locator } from "@playwright/test";

export class OnboardingPage {
  readonly page: Page;
  readonly projectNameInput: Locator;
  readonly websiteUrlInput: Locator;
  readonly createButton: Locator;
  readonly recommendedTiles: Locator;

  constructor(page: Page) {
    this.page = page;
    this.projectNameInput = page.locator("#name, input[name=name]");
    this.websiteUrlInput = page.locator("#website_url, input[name=website_url]");
    this.createButton = page.locator("button[type=submit], button:has-text('Create')");
    this.recommendedTiles = page.locator("[data-testid=recommended-tile], [class*=tile]");
  }

  async goto() {
    await this.page.goto("/onboarding");
  }

  async createProject(name: string, websiteUrl?: string) {
    await this.projectNameInput.fill(name);
    if (websiteUrl) await this.websiteUrlInput.fill(websiteUrl);
    await this.createButton.click();
    await this.page.waitForLoadState("networkidle");
  }

  async connectTile(mcpKey: string) {
    await this.page.locator(`[data-mcp-key="${mcpKey}"] button`).click();
    await this.page.waitForLoadState("networkidle");
  }
}

**Files:**

**Interfaces:**
- Consumes: everything above
- Produces: CI pipeline that runs E2E tests on push/PR

- [ ] **Step 1: Create `.github/workflows/e2e.yml`**

```yaml
name: E2E Tests

on:
  push:
    branches: [developer]
    paths:
      - "notfair-cmo/src/**"
      - "notfair-cmo/e2e/**"
      - "notfair-cmo/package.json"
      - ".github/workflows/e2e.yml"
  pull_request:
    paths:
      - "notfair-cmo/src/**"
      - "notfair-cmo/e2e/**"
      - "notfair-cmo/package.json"
      - ".github/workflows/e2e.yml"

env:
  E2E_TEST: "1"
  # Use SQLite (no external DB needed)
  PORT: 3326

defaults:
  run:
    working-directory: notfair-cmo

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm
          cache-dependency-path: notfair-cmo/pnpm-lock.yaml

      - run: pnpm install --frozen-lockfile

      - run: npx playwright install chromium

      - name: Start dev server
        run: |
          pnpm dev &
          npx wait-on http://localhost:3326 --timeout 30000

      - name: Run E2E tests
        run: npx playwright test --config=e2e/playwright.config.ts

      - uses: actions/upload-artifact@v4
        if: failure()
        with:
          name: playwright-report
          path: notfair-cmo/playwright-report/
          retention-days: 7
```
